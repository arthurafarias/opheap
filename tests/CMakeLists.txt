# Each header-only library carries its own tests colocated with the classes they cover,
# under its "include/.../testing/*_test.hpp" tree. Every such header self-registers a
# test_group and is discovered here regardless of which library it lives in.
set(OPHEAP_TEST_INCLUDE_ROOTS
    "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/libopheap-core/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/libopheap-module-cli/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/libopheap-module-sql/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/libopheap-utils-serialization-json/include"
)

set(GENERATED_TESTS "${CMAKE_CURRENT_BINARY_DIR}/all_tests.cpp")
file(WRITE "${GENERATED_TESTS}" "// Generated by CMake. Every testing/*_test.hpp header self-registers a test_group.\n")
foreach(INCLUDE_ROOT IN LISTS OPHEAP_TEST_INCLUDE_ROOTS)
    file(GLOB_RECURSE ROOT_TEST_HEADERS CONFIGURE_DEPENDS "${INCLUDE_ROOT}/*_test.hpp")
    foreach(TEST_HEADER IN LISTS ROOT_TEST_HEADERS)
        file(RELATIVE_PATH RELATIVE_HEADER "${INCLUDE_ROOT}" "${TEST_HEADER}")
        file(APPEND "${GENERATED_TESTS}" "#include <${RELATIVE_HEADER}>\n")
    endforeach()
endforeach()

add_executable(opheap_tests main.cpp "${GENERATED_TESTS}")
target_link_libraries(opheap_tests PRIVATE opheap::opheap opheap::module::sql opheap::module::cli opheap::json)
target_compile_features(opheap_tests PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(opheap_tests PRIVATE /W4 /permissive-)
else()
    target_compile_options(opheap_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME opheap_tests COMMAND opheap_tests)
